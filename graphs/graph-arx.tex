\begin{tikzpicture}[every node/.style={font=\small}]

    %==========================================================
    % One slice of the Bayesian ARX model with M_y = 1,
    %   y_k ~ N(A^T phi(u_k, y_{k-1}), W^{-1}),  Theta = (A, W).
    % The state is the output buffer, so the transition f_x is the
    % composite of the feature map phi and the likelihood f_y.
    %==========================================================

    % parameter chain (Theta = (A, W))
    \node [style=deterministic] (eqT) at (0.7, 1.9) {$=$};
    \draw (-1.9, 1.9) -- (eqT) -- (2.8, 1.9);
    \node [style=edgelabel, above] at (-1.0, 1.9) {$\Theta$};

    % transition: features and likelihood
    \node [style=deterministic] (phi) at (-0.5, 0) {$\phi$};
    \node [draw, rectangle, minimum width=7mm, minimum height=6mm] (fy) at (0.7, 0) {$f_y$};
    \node [style=stochastic] (fu) at (-0.5, 1.25) {$f_u$};
    \draw (eqT) -- (fy);
    \draw (fu) -- (phi);
    \draw (phi) -- (fy);
    \node [style=edgelabel, right=0.5mm] at ($(fu.south)!0.4!(phi.north)$) {$u_k$};
    \node [style=edgelabel, above] at ($(phi)!0.5!(fy)$) {$\phi_k$};

    % output chain and goal prior
    \node [style=deterministic] (eqy) at (2.2, 0)    {$=$};
    \node [style=stochastic]    (g)   at (2.2, -1.0) {$f_*$};
    \draw (-1.9, 0) -- (phi);
    \draw (fy) -- (eqy) -- (2.8, 0);
    \draw (eqy) -- (g);
    \node [style=edgelabel, above] at (-1.3, 0) {$y_{k-1}$};
    \node [style=edgelabel, above] at ($(fy.east)!0.55!(eqy.west)$) {$y_k$};

    % dashed box around the composite transition node f_x
    \draw [dashed, rounded corners=2pt] (-0.8, -0.45) rectangle (1.15, 0.45);
    \node [style=edgelabel] at (-0.95, -0.6) {$f_x$};

\end{tikzpicture}
